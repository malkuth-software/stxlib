# Install our PID 1 (runtime/init/stxinit.cpp) and make the kernel start it
# instead of /sbin/init (loader variable init_path, sys/kern/init_main.c:759-762).
# After this, the VM boots to stxinit's marker and has no login: run it last.
fetch -q -o /sbin/stxinit {http}/stxinit
test "$(sha256 -q /sbin/stxinit)" = "{init_sha256}"
chmod 555 /sbin/stxinit
sed -i '' '/^init_path=/d' /boot/loader.conf.local
printf '%s\n' 'init_path="/sbin/stxinit"' >> /boot/loader.conf.local
cat /boot/loader.conf.local
sync
